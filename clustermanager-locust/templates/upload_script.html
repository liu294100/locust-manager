    脚本编写指南
    基本要求
    • 文件必须是 .py 格式
    • 包含 Locust 用户类
    • 继承自 HttpUser 或 User
    • 定义 task 方法
    最佳实践
    • 设置合适的 wait_time
    • 使用有意义的任务名称
    • 添加适当的错误处理
    • 包含必要的注释
    示例脚本
    from locust import HttpUser, task, between

class WebsiteUser(HttpUser):
    wait_time = between(1, 3)
    
    def on_start(self):
        """用户开始时执行"""
        pass
    
    @task(3)
    def view_homepage(self):
        """访问首页"""
        self.client.get("/")
    
    @task(1)
    def view_about(self):
        """访问关于页面"""
        self.client.get("/about")
